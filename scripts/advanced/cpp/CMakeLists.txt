cmake_minimum_required(VERSION 3.16)
project(linux_advanced_units LANGUAGES CXX)
add_library(units units.cpp)
target_include_directories(units PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(units PUBLIC cxx_std_17)
add_executable(angle_advanced main.cpp)
target_link_libraries(angle_advanced PRIVATE units)
enable_testing()
add_executable(test_units test_units.cpp)
target_link_libraries(test_units PRIVATE units)
add_test(NAME angle_conversion COMMAND test_units)
